function(konveyor_add_module name)
    file(GLOB_RECURSE sources CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/${name}/*.cpp)
    if(NOT sources)
        add_library(konveyor_${name} INTERFACE)
        target_include_directories(konveyor_${name} INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
        target_link_libraries(konveyor_${name} INTERFACE Qt6::Core Qt6::Gui ${ARGN})
        return()
    endif()
    add_library(konveyor_${name} STATIC ${sources})
    target_include_directories(konveyor_${name} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(konveyor_${name} PUBLIC Qt6::Core Qt6::Gui ${ARGN})
    target_compile_options(konveyor_${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    set_target_properties(konveyor_${name} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endfunction()

function(konveyor_compose_shader out)
    set(text "")
    foreach(file IN LISTS ARGN)
        file(READ "${file}" part)
        string(APPEND text "${part}")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
    endforeach()
    set(${out} "${text}" PARENT_SCOPE)
endfunction()
