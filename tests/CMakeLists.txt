function(konveyor_add_unit_test source module)
    get_filename_component(name ${source} NAME_WE)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE konveyor_${module} Qt6::Test)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror)
    target_compile_definitions(${name} PRIVATE KONVEYOR_SOURCE_DIR="${PROJECT_SOURCE_DIR}")
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES LABELS "unit;${module}" ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endfunction()

foreach(module kdl anim config render layout ipc)
    file(GLOB_RECURSE module_tests CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/unit/${module}/test_*.cpp)
    foreach(test_source ${module_tests})
        konveyor_add_unit_test(${test_source} ${module})
    endforeach()
endforeach()

file(GLOB KONVEYOR_NESTED_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/nested/test_*.py)
foreach(test_script ${KONVEYOR_NESTED_TESTS})
    get_filename_component(name ${test_script} NAME_WE)
    add_test(NAME nested_${name} COMMAND python3 ${test_script})
    set_tests_properties(nested_${name} PROPERTIES LABELS nested ENVIRONMENT "KONVEYOR_BUILD_DIR=${PROJECT_BINARY_DIR}" TIMEOUT 600)
endforeach()
